export function AppProviders({ children }: { children: ReactNode }) {
  const queryClient = useMemo(
    () =>
      new QueryClient({
        defaultOptions: {
          queries: {
            staleTime: 20_000,
            retry: 1,
            refetchOnWindowFocus: false,
          },
          mutations: { retry: 0 },
        },
      }),
    [],
  );

  const [user, setUser] = useState<SessionUser | null>(null);
  const [authLoading, setAuthLoading] = useState(true);
  const [theme, setThemeState] = useState<Theme>("dark");
  const [locale, setLocaleState] = useState<string>("ru");
  const [toasts, setToasts] = useState<Toast[]>([]);
  const [online, setOnline] = useState(true);

  const loadSession = useCallback(async () => {
    setAuthLoading(true);
    try {
      const res = await fetch("/api/auth", { cache: "no-store" });
      if (res.ok) {
        const data = (await res.json()) as { user: SessionUser | null };
        setUser(data.user);
        if (data.user) {
          setThemeState((data.user.theme as Theme) ?? "system");
          setLocaleState(data.user.locale ?? "ru");
          return;
        }
      }
      setUser(null);
    } catch {
      setUser(null);
    } finally {
      setAuthLoading(false);
    }
  }, []);

  useEffect(() => {
    void loadSession();
  }, [loadSession]);

  // Local overrides are applied before the session arrives to avoid a flash.
  useEffect(() => {
    const storedTheme = localStorage.getItem(THEME_KEY) as Theme | null;
    const storedLocale = localStorage.getItem(LOCALE_KEY);
    if (!storedLocale) localStorage.setItem(LOCALE_KEY, detectBrowserLocale());
    if (storedTheme) setThemeState(storedTheme);
    setLocaleState(storedLocale ?? detectBrowserLocale());
    setOnline(navigator.onLine);
    const up = () => setOnline(true);
    const down = () => setOnline(false);
    window.addEventListener("online", up);
    window.addEventListener("offline", down);
    return () => {
      window.removeEventListener("online", up);
      window.removeEventListener("offline", down);
    };
  }, []);

  useEffect(() => {
    document.documentElement.dataset.theme = theme;
    localStorage.setItem(THEME_KEY, theme);
  }, [theme]);

  useEffect(() => {
    document.documentElement.lang = locale;
    localStorage.setItem(LOCALE_KEY, locale);
  }, [locale]);

  const toast = useCallback((text: string, tone: Toast["tone"] = "info") => {
    const id = Math.random().toString(36).slice(2);
    setToasts((prev) => [...prev.slice(-2), { id, text, tone }]);
    window.setTimeout(() => setToasts((prev) => prev.filter((t) => t.id !== id)), 3600);
  }, []);

  const value = useMemo<Ctx>(
    () => ({
      user,
      setUser,
      authLoading,
      refreshUser: loadSession,
      locale,
      setLocale: setLocaleState,
      theme,
      setTheme: setThemeState,
      t: (key, vars) => translate(locale, key, vars),
      fmt: {

        date: (d, opts) => formatDate(d, locale, opts),
        time: (d) => formatTime(d, locale),
        relative: (d) => formatRelative(d, locale),
        duration: (ms) => formatDuration(ms),
        bytes: (b) => formatBytes(b, locale),
      },
      toast,
      toasts,
      online,
      queryClient,
    }),
    [user, authLoading, loadSession, locale, theme, toast, toasts, online, queryClient],
  );

  return (
    <QueryClientProvider client={queryClient}>
      <AppCtx.Provider value={value}>
        {children}
        <ToastHost />
      </AppCtx.Provider>

  );
}

import { z } from "zod";

export class ClientApiError extends Error {
  constructor(
    public status: number,
    public code: string,
    message: string,
  ) {
    super(message);
  }
}

const errorShape = z.object({ error: z.object({ code: z.string(), message: z.string().optional() }) });

async function parseResponse<T>(res: Response): Promise<T> {
  const text = await res.text();
  let data: unknown = null;
  if (text) {
    try {
      data = JSON.parse(text);
    } catch {
      data = null;
    }
  }
  if (!res.ok) {
    const parsed = errorShape.safeParse(data);
    throw new ClientApiError(
      res.status,
      parsed.success ? parsed.data.error.code : "http_error",
      parsed.success ? (parsed.data.error.message ?? parsed.data.error.code) : `Request failed (${res.status})`,
    );
  }
  return data as T;
}

export async function api<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(path, {
    ...init,
    headers: {
      ...(init?.body && !(init.body instanceof FormData) ? { "Content-Type": "application/json" } : {}),
      ...(init?.headers ?? {}),
    },
    cache: "no-store",
  });
  return parseResponse<T>(res);
}

export const post = <T>(path: string, body?: unknown) =>
  api<T>(path, { method: "POST", body: body === undefined ? undefined : JSON.stringify(body) });
export const patch = <T>(path: string, body?: unknown) =>
  api<T>(path, { method: "PATCH", body: body === undefined ? undefined : JSON.stringify(body) });
export const del = <T>(path: string, body?: unknown) =>
  api<T>(path, { method: "DELETE", body: body === undefined ? undefined : JSON.stringify(body) });
export const get = <T>(path: string) => api<T>(path, { method: "GET" });

/** Загрузка напрямую в S3 (Selectel) через presigned URL. */
export async function uploadMedia(
  file: File,
  opts: {
    purpose: "image" | "video" | "audio" | "file" | "avatar" | "cover" | "story" | "clip";
    visibility?: string;
    onProgress?: (percent: number) => void;
    signal?: AbortSignal;
  },
): Promise<{ media: { id: string; key: string; url: string; kind: string; mime: string; size: number } }> {
  // 1. Определяем расширение по MIME
  const mime = file.type.toLowerCase().split(";")[0].trim();
  const extMap: Record<string, string> = {
    "image/jpeg": "jpg", "image/png": "png", "image/webp": "webp", "image/gif": "gif", "image/avif": "avif",
    "video/mp4": "mp4", "video/webm": "webm", "video/quicktime": "mov",
    "audio/mpeg": "mp3", "audio/webm": "weba", "audio/ogg": "ogg", "audio/mp4": "m4a", "audio/wav": "wav",
    "application/pdf": "pdf", "text/plain": "txt", "application/zip": "zip",
  };
  const ext = extMap[mime] ?? "bin";

  // 2. Получаем presigned URL от нашего API
  const { presignedUrl, key, mediaId } = await post<{ presignedUrl: string; key: string; mediaId: string }>(
    "/api/upload/presigned",
    { contentType: mime, ext, purpose: opts.purpose, visibility: opts.visibility }
  );

  // 3. Загружаем файл напрямую в S3 (с прогрессом)
  return new Promise((resolve, reject) => {
    const xhr = new XMLHttpRequest();
    xhr.open("PUT", presignedUrl);
    xhr.setRequestHeader("Content-Type", mime);
    
    xhr.upload.onprogress = (event) => {
      if (event.lengthComputable) opts.onProgress?.(Math.round((event.loaded / event.total) * 100));
    };

    xhr.onload = async () => {
      if (xhr.status >= 200 && xhr.status < 300) {
        // Файл загружен, возвращаем данные
        resolve({
          media: {
            id: mediaId,
            key,
            url: `/api/media/${key}`,
            kind: opts.purpose === "image" ? "image" : opts.purpose === "video" || opts.purpose === "clip" ? "video" : "file",
            mime,
            size: file.size,
          },
        });
      } else {
        reject(new ClientApiError(xhr.status, "upload_failed", "Upload to S3 failed"));
      }
    };

    xhr.onerror = () => reject(new ClientApiError(0, "network_error", "Network error during upload"));
    xhr.onabort = () => reject(new ClientApiError(0, "aborted", "Upload cancelled"));
    opts.signal?.addEventListener("abort", () => xhr.abort());
    xhr.send(file);
  });
}

export function StoryViewer({ groups, startIndex, onClose }: { groups: TrayGroup[]; startIndex: number; onClose: () => void }) {
  const { t, fmt, toast } = useApp();
  const queryClient = useQueryClient();
  const [groupIndex, setGroupIndex] = useState(startIndex);
  const [storyIndex, setStoryIndex] = useState(0);
  const [progress, setProgress] = useState(0);
  const [paused, setPaused] = useState(false);
  const [viewers, setViewers] = useState<{ userId: string; username: string; displayName: string; viewedAt: string }[] | null>(null);
  const [reportOpen, setReportOpen] = useState(false);
  const timer = useRef<number | null>(null);

  const group = groups[groupIndex];
  const story = group?.stories[storyIndex];

  const markViewed = useMutation({
    mutationFn: (id: string) => postApi(`/api/stories/${id}`, { action: "view" }),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: qk.storyTray }),
  });
  const deleteStory = useMutation({
    mutationFn: (id: string) => del(`/api/stories/${id}`),
    onSuccess: () => {
      toast(t("story.delete"), "success");
      queryClient.invalidateQueries({ queryKey: qk.storyTray });
      next();
    },
  });

  const next = () => {
    setProgress(0);
    if (!group) return onClose();
    if (storyIndex + 1 < group.stories.length) return setStoryIndex(storyIndex + 1);
    if (groupIndex + 1 < groups.length) {
      setStoryIndex(0);
      return setGroupIndex(groupIndex + 1);
    }
    onClose();
  };
  const prev = () => {
    setProgress(0);
    if (storyIndex > 0) return setStoryIndex(storyIndex - 1);
    if (groupIndex > 0) {
      setStoryIndex(0);
      setGroupIndex(groupIndex - 1);
    }
  };

  useEffect(() => {
    if (!story) return;
    markViewed.mutate(story.id);
    const duration = story.mediaKind === "video" ? 15_000 : 6_000;
    const started = Date.now();
    const tick = () => {
      if (paused) return;
      const value = Math.min(100, ((Date.now() - started) / duration) * 100);
      setProgress(value);
      if (value >= 100) next();
    };
    timer.current = window.setInterval(tick, 100);
    return () => {
      if (timer.current) window.clearInterval(timer.current);
    };
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [story?.id, paused]);

  useEffect(() => {
    const onKey = (event: KeyboardEvent) => {
      if (event.key === "Escape") onClose();
      if (event.key === "ArrowRight") next();
      if (event.key === "ArrowLeft") prev();
    };
    window.addEventListener("keydown", onKey);
    return () => window.removeEventListener("keydown", onKey);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [groupIndex, storyIndex, groups.length]);

  useEffect(() => {
    if (!story?.isMine) {
      setViewers(null);
      return;
    }
    void get<{ viewers?: { userId: string; username: string; displayName: string; viewedAt: string }[] }>(`/api/stories/${story.id}`).then((data) =>
      setViewers(data.viewers ?? []),
    );
  }, [story?.id, story?.isMine]);

  if (!story) return null;

  return (
    <div className="fixed inset-0 z-[95] flex items-center justify-center bg-black/95" role="dialog" aria-modal="true" aria-label={t("feed.stories")}>
      <button className="absolute inset-0 h-full w-full cursor-default" onClick={onClose} aria-label={t("common.close")} tabIndex={-1} />
      <div className="relative flex h-full w-full max-w-md flex-col md:h-[92vh] md:rounded-3xl md:border md:border-line">
        <div className="absolute inset-x-0 top-0 z-10 space-y-2 p-3">
          <div className="flex gap-1">
            {group!.stories.map((_, index) => (
              <Progress key={index} value={index < storyIndex ? 100 : index === storyIndex ? progress : 0} className="h-0.5 flex-1 bg-white/25" />
            ))}
          </div>
          <div className="flex items-center gap-2">
            <Link href={`/u/${story.author.username}`} onClick={onClose}>
              <Avatar name={story.author.displayName} url={story.author.avatarUrl} size={32} />
            </Link>
            <div className="min-w-0 flex-1">
              <p className="truncate text-xs font-semibold text-white">{story.author.displayName}</p>
              <p className="text-[10px] text-white/60">{fmt.relative(story.createdAt)}</p>
            </div>
            {story.isMine ? (
              <IconButton label={t("story.delete")} className="text-white hover:bg-white/10" onClick={() => deleteStory.mutate(story.id)}>
                <Trash2 className="h-4 w-4" />
              </IconButton>
            ) : (
              <IconButton label={t("common.report")} className="text-white hover:bg-white/10" onClick={() => setReportOpen(true)}>
                <Flag className="h-4 w-4" />
              </IconButton>
            )}

export function PostCard({ post, source = "feed" }: { post: PostPayload; source?: string }) {
  const { t, fmt, user, toast, locale } = useApp();
  const queryClient = useQueryClient();
  const [showComments, setShowComments] = useState(false);
  const [reportOpen, setReportOpen] = useState(false);

  const likeMutation = useMutation({
    mutationFn: () => postApi<{ liked: boolean; likeCount: number }>(`/api/posts/${post.id}/actions`, { action: "like", source }),
    onMutate: async () => {
      await queryClient.cancelQueries({ queryKey: ["feed"] });
      return null;
    },
    onSuccess: (data) => {
      queryClient.setQueriesData<{ posts?: PostPayload[] }>({}, (old) => {
        if (!old?.posts) return old;
        return {
          ...old,
          posts: old.posts.map((p) => (p.id === post.id ? { ...p, likedByViewer: data.liked, likeCount: data.likeCount } : p)),
        };
      });
    },
    onError: () => toast(t("common.error"), "error"),
  });

  const saveMutation = useMutation({
    mutationFn: () => postApi(`/api/posts/${post.id}/actions`, { action: post.savedByViewer ? "unsave" : "save" }),
    onSuccess: () => {
      toast(post.savedByViewer ? t("common.saved") : t("post.saved"), "success");
      queryClient.invalidateQueries({ queryKey: qk.saved });
    },
  });

  const deleteMutation = useMutation({
    mutationFn: () => del(`/api/posts/${post.id}`),
    onSuccess: () => {
      toast(t("post.deleted"), "success");
      queryClient.invalidateQueries({ queryKey: ["feed"] });
      queryClient.invalidateQueries({ queryKey: ["profile"] });
    },
  });

  const blockMutation = useMutation({
    mutationFn: (action: "block") => postApi(`/api/users/${post.author.username}/actions`, { action }),
    onSuccess: () => {
      toast(t("common.block"), "success");
      queryClient.invalidateQueries({ queryKey: ["feed"] });
    },
  });

  const comments = useQuery({
    queryKey: qk.comments("post", post.id),
    queryFn: () => get<{ comments: CommentRow[] }>(`/api/comments?entityType=post&entityId=${post.id}`),
    enabled: showComments,
  });

  return (
    <article className="animate-fade-up rounded-2xl border border-line bg-surface">
      <header className="flex items-start gap-3 px-4 pt-3.5">
        <Link href={`/u/${post.author.username}`} className="shrink-0">
          <Avatar name={post.author.displayName} url={post.author.avatarUrl} size={40} />
        </Link>
        <div className="min-w-0 flex-1">
          <div className="flex items-center gap-1.5">
            <Link href={`/u/${post.author.username}`} className="truncate text-sm font-semibold text-ink hover:underline">
              {post.author.displayName}
            </Link>
            {post.author.isCreator && <span className="text-[10px] text-brand">✦</span>}
          </div>
          <p className="truncate text-[11px] text-muted">
            @{post.author.username} · {fmt.relative(post.createdAt)}
            {post.audience !== "public" && ` · ${t(`settings.audience${post.audience === "followers" ? "Followers" : post.audience === "contacts" ? "Contacts" : "Private"}` as never)}`}
          </p>
        </div>
        {post.conversation && (
          <Link href={`/c/${post.conversation.slug ?? post.conversation.id}`}>
            <span className="rounded-full border border-line px-2 py-0.5 text-[10px] text-muted hover:text-brand">
              {post.conversation.title}
            </span>
          </Link>
        )}
        <Menu
          trigger={({ toggle }) => (
            <IconButton label={t("common.edit")} onClick={toggle}>
              <MoreHorizontal className="h-4 w-4" />
            </IconButton>
          )}
        >
          <MenuItem icon={<Flag className="h-3.5 w-3.5" />} onClick={() => setReportOpen(true)}>
            {t("common.report")}
          </MenuItem>
          {post.author.id !== user?.id && (
            <MenuItem icon={<Ban className="h-3.5 w-3.5" />} tone="danger" onClick={() => blockMutation.mutate("block")}>
              {t("common.block")} @{post.author.username}
            </MenuItem>
          )}
          {post.isMine && (
            <MenuItem icon={<Trash2 className="h-3.5 w-3.5" />} tone="danger" onClick={() => deleteMutation.mutate()}>
              {t("common.delete")}
            </MenuItem>
          )}
        </Menu>
      </header>

      {post.text && <p className="whitespace-pre-wrap px-4 py-3 text-sm leading-relaxed text-ink">{post.text}</p>}

      {post.media.length > 0 && (
        <div className={cn("grid gap-0.5 overflow-hidden rounded-t-none", post.media.length > 1 ? "grid-cols-2" : "grid-cols-1")}>
          {post.media.slice(0, 4).map((m) =>
            m.kind === "video" ? (
              <Link key={m.id} href={post.video ? `/videos/${post.video.id}` : "/clips"} className="relative block bg-black">
                <video src={m.url} muted playsInline preload="metadata" className="max-h-[420px] w-full object-cover" />

              </Link>
            ) : (
              // eslint-disable-next-line @next/next/no-img-element
              <img
                key={m.id}
                src={m.url}
                alt={post.text?.slice(0, 80) ?? "UZON media"}
                loading="lazy"
                className="max-h-[520px] w-full bg-surface2 object-cover"
              />
            ),
          )}
        </div>
      )}

      {post.video && post.media.length === 0 && (
        <Link href={post.video.kind === "clip" ? "/clips" : `/videos/${post.video.id}`} className="mx-4 mb-2 flex items-center gap-3 rounded-xl border border-line bg-surface2 p-3">
          <PlayCircle className="h-8 w-8 text-brand" />
          <span className="min-w-0">
            <span className="block truncate text-xs font-semibold text-ink">{post.video.title}</span>
            <span className="block text-[11px] text-muted">
              {fmt.number(post.video.viewCount)} {t("video.views")} · {fmt.duration(post.video.durationMs)}

        </Link>
      )}

      <div className="flex items-center gap-1 px-2 py-2">
        <ActionButton
          active={post.likedByViewer}
          label={t("post.comment")}
          onClick={() => likeMutation.mutate()}
          pending={likeMutation.isPending}
          icon={<Heart className={cn("h-[18px] w-[18px]", post.likedByViewer && "fill-danger text-danger")} />}
          text={fmt.number(post.likeCount)}
        />
        <ActionButton
          label={t("post.comments")}
          onClick={() => setShowComments((v) => !v)}
          icon={<MessageCircle className="h-[18px] w-[18px]" />}
          text={fmt.number(post.commentCount)}
        />
        <ActionButton
          label={t("common.save")}
          active={post.savedByViewer}
          onClick={() => saveMutation.mutate()}
          icon={<Bookmark className={cn("h-[18px] w-[18px]", post.savedByViewer && "fill-brand text-brand")} />}
        />
        <ActionButton
          label={t("common.share")}
          onClick={() => {
            const url = `${window.location.origin}/post/${post.id}`;
            void navigator.clipboard?.writeText(url);
            toast(t("common.copied"), "success");
          }}
          icon={<Send className="h-[18px] w-[18px]" />}
        />
      </div>

      {showComments && (
        <div className="border-t border-line px-4 py-3">
          {comments.isLoading && <Skeleton className="h-16 w-full" />}
          {comments.isError && <ErrorState message={t("common.error")} onRetry={() => comments.refetch()} />}
          {comments.data && comments.data.comments.length === 0 && (
            <p className="py-2 text-xs text-muted">{t("post.noComments")}</p>
          )}
          <div className="space-y-3">
            {comments.data?.comments.map((comment) => (
              <div key={comment.id} className="flex gap-2.5">
                <Avatar name={comment.author.displayName} url={comment.author.avatarUrl} size={30} />
                <div className="min-w-0 flex-1 rounded-xl bg-surface2 px-3 py-2">
                  <p className="text-xs font-semibold text-ink">
                    <Link href={`/u/${comment.author.username}`} className="hover:underline">
                      {comment.author.displayName}
                    </Link>
                    <span className="ml-2 font-normal text-muted">{fmt.relative(comment.createdAt)}</span>
                  </p>

function sniff(buffer: Buffer, declaredMime: string): boolean {
  if (buffer.length < 12) return false;
  const isJpeg = buffer[0] === 0xff && buffer[1] === 0xd8 && buffer[2] === 0xff;
  const isPng = buffer.subarray(0, 8).equals(Buffer.from([0x89, 0x50, 0x4e, 0x47, 0x0d, 0x0a, 0x1a, 0x0a]));
  const isWebm = buffer.subarray(0, 4).equals(Buffer.from([0x1a, 0x45, 0xdf, 0xa3]));
  const isGif = buffer.subarray(0, 3).toString("latin1") === "GIF";
  const isRiff = buffer.subarray(0, 4).toString("latin1") === "RIFF";
  const isMp4 = buffer.subarray(4, 8).toString("latin1") === "ftyp";
  const isMov = isMp4;
  const isPdf = buffer.subarray(0, 5).toString("latin1") === "%PDF-";
  const isZip = buffer[0] === 0x50 && buffer[1] === 0x4b;
  const isId3 = buffer.subarray(0, 3).toString("latin1") === "ID3";
  const isOgg = buffer.subarray(0, 4).toString("latin1") === "OggS";
  const detected = isJpeg || isPng || isWebm || isGif || isRiff || isMp4 || isMov || isPdf || isZip || isId3 || isOgg;
  if (!detected) return false;
  const kind = mediaKind(declaredMime);
  if (kind === "image") return isJpeg || isPng || isWebm || isGif || (isRiff && declaredMime === "image/webp");
  if (kind === "video") return isMp4 || isWebm || isMov;
  if (kind === "audio") return isId3 || isOgg || isWebm || isMp4 || isRiff;
  return isPdf || isZip;
}

export async function writeMedia(opts: {
  file: File;
  ownerId: string;
  maxBytes: number;
  allowed?: string[];
}): Promise<{ key: string; size: number; mime: string; kind: ReturnType<typeof mediaKind> }> {
  const { file, ownerId, maxBytes } = opts;
  const allowed = opts.allowed ?? ALLOWED_MIME;

  // ИСПРАВЛЕНИЕ: если file.type пустой, определяем MIME по расширению
  let mime = file.type.toLowerCase().split(";")[0].trim();
  if (!mime || mime === "application/octet-stream") {
    const fromExt = mimeFromExt(file.name);
    if (fromExt) mime = fromExt;
  }

  if (!allowed.includes(mime)) {
    throw new ApiError(415, "unsupported_media_type", `Unsupported file type: ${mime || "unknown"}`);
  }
  if (file.size > maxBytes) {
    throw new ApiError(413, "file_too_large", `File exceeds the ${Math.round(maxBytes / 1024 / 1024)} MB limit`);
  }
  if (file.size === 0) throw new ApiError(422, "empty_file", "The selected file is empty");

  const buffer = Buffer.from(await file.arrayBuffer());
  if (!sniff(buffer, mime)) {
    throw new ApiError(415, "content_mismatch", "File content does not match its declared type");
  }

  const ext = EXT[mime] ?? "bin";
  const key = `${ownerId.slice(0, 8)}-${randomUUID()}.${ext}`;

  await s3Client.send(
    new PutObjectCommand({
      Bucket: S3_BUCKET,
      Key: key,
      Body: buffer,
      ContentType: mime,
      ACL: "public-read",
    }),
  );

  return { key, size: file.size, mime, kind: mediaKind(mime) };
}

export async function streamMedia(opts: {
  key: string;
  mime: string;
  rangeHeader: string | null;
  download?: boolean;
}): Promise<Response | null> {
  const file = await readMediaFile(opts.key);
  if (!file) return null;
  const total = file.size;
  const headers = new Headers({
    "Accept-Ranges": "bytes",
    "Content-Type": opts.mime,
    "Cache-Control": "private, max-age=3600",
  });
  if (opts.download) headers.set("Content-Disposition", `attachment; filename="${opts.key.split("/").pop()}"`);

  const range = opts.rangeHeader?.match(/bytes=(\d*)-(\d*)/);
  if (range && (range[1] || range[2])) {
    const start = range[1] ? Number(range[1]) : Math.max(0, total - Number(range[2]));
    const end = range[1] && range[2] ? Math.min(total - 1, Number(range[2])) : total - 1;
    if (start >= total || end < start) {
      return new Response(null, { status: 416, headers: { "Content-Range": `bytes */${total}` } });
    }
    headers.set("Content-Range", `bytes ${start}-${end}/${total}`);
    headers.set("Content-Length", String(end - start + 1));
    return new Response(nodeStreamToWeb(createReadStream(file.path, { start, end })), { status: 206, headers });
  }

  headers.set("Content-Length", String(total));
  return new Response(nodeStreamToWeb(createReadStream(file.path)), { status: 200, headers });
}

function nodeStreamToWeb(stream: NodeJS.ReadableStream): ReadableStream<Uint8Array> {
  return new ReadableStream<Uint8Array>({
    start(controller) {
      stream.on("data", (chunk: Buffer) => controller.enqueue(new Uint8Array(chunk)));
      stream.on("end", () => controller.close());
      stream.on("error", (err) => controller.error(err));
    },
    cancel() {
      (stream as unknown as { destroy?: () => void }).destroy?.();
    },
  });
}

    .select({ audience: videos.audience, ownerId: videos.ownerId, deletedAt: videos.deletedAt })
    .from(videos)
    .where(eq(videos.mediaId, row.id))
    .limit(1);
  if (videoRef) {
    if (videoRef.deletedAt) throw new ApiError(410, "gone", "Video removed");
    await assertAudience(videoRef.audience, videoRef.ownerId, viewerId);
    return;
  }

  const [postRef] = await db
    .select({ audience: posts.audience, authorId: posts.authorId, deletedAt: posts.deletedAt })
    .from(postMedia)
    .innerJoin(posts, eq(posts.id, postMedia.postId))
    .where(eq(postMedia.mediaId, row.id))
    .limit(1);
  if (postRef) {
    if (postRef.deletedAt) throw new ApiError(410, "gone", "Post removed");
    await assertAudience(postRef.audience, postRef.authorId, viewerId);
    return;
  }

  if (row.visibility === "public") return;
  await assertAudience(row.visibility, row.ownerId, viewerId);
}

async function assertAudience(audience: string, ownerId: string, viewerId: string | null) {
  if (audience === "public") return;
  if (!viewerId) throw new ApiError(403, "forbidden", "Sign in to view this content");
  const ctx = await getRelation(viewerId, ownerId);
  if (ctx.blocked) throw new ApiError(403, "blocked");
  const allowed = evaluatePrivacyRule(audience as PrivacyRule, ctx);
  if (!allowed) throw new ApiError(403, "forbidden", "Restricted by the author's privacy settings");
}

const CACHE_NAME = "uzon-pwa-v2";

self.addEventListener("install", (event) => {
  event.waitUntil(
    caches.open(CACHE_NAME).then((cache) => {
      return cache.addAll([
        "/",
        "/manifest.webmanifest",
      ]);
    })
  );

  self.skipWaiting();
});

self.addEventListener("activate", (event) => {
  event.waitUntil(
    caches.keys().then((keys) =>
      Promise.all(
        keys
          .filter(
            (key) => key.startsWith("uzon-pwa-") && key !== CACHE_NAME
          )
          .map((key) => caches.delete(key))
      )
    ).then(() => self.clients.claim())
  );
});

self.addEventListener("fetch", (event) => {
  const request = event.request;

  if (
    request.method !== "GET" ||
    new URL(request.url).origin !== self.location.origin
  ) {
    return;
  }

  event.respondWith(
    fetch(request)
      .then((response) => {
        if (response.ok) {
          const copy = response.clone();

          caches.open(CACHE_NAME).then((cache) => {
            cache.put(request, copy);
          });
        }

        return response;
      })
      .catch(async () => {
        const cached = await caches.match(request);

        if (cached) return cached;

        if (request.mode === "navigate") {
          return (
            (await caches.match("/")) ||
            new Response("UZON недоступен без подключения к интернету.", {
              status: 503,
              headers: { "Content-Type": "text/plain; charset=utf-8" },
            })
          );
        }

        return new Response("Нет подключения к интернету", {
          status: 503,
          headers: { "Content-Type": "text/plain; charset=utf-8" },
        });
      })
  );
});
